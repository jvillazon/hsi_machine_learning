            Number of columns in the legend.

            loc='center',
            frameon=True,
            fontsize=fontsize,
            title=title,
            title_fontsize=title_fontsize,
            ncol=ncol,
        )
        legend.get_frame().set_facecolor('white')
        legend.get_frame().set_alpha(0.95)
        ax.axis('off')
        fig.savefig(save_path, dpi=300, bbox_inches='tight', facecolor='white')
        plt.close(fig)
